// Smart error detection and solution matching
class ErrorResolver {
    constructor() {
        this.patterns = new ErrorPatternDatabase();
        this.solutions = new SolutionProvider();
    }
    
    async resolveError(errorText, context) {
        const pattern = await this.patterns.findMatch(errorText);
        if (pattern) {
            const solutions = await this.solutions.getSolutions(pattern, context);
            return {
                pattern: pattern.name,
                solutions: solutions,
                confidence: pattern.confidence,
                quickFix: solutions[0]?.command
            };
        }
        return this.getFallbackSuggestions(errorText);
    }
}
